import java.util.*;
import java.lang.Math;

public class JavaApplication {
    private static Scanner input = new Scanner(System.in);

    public static void main(String[] args) {
        int n, i, b, d;

        n = Integer.parseInt(input.nextLine());
        while (n > 8 || n < 1) {
            n = Integer.parseInt(input.nextLine());
        }
        String[] nm = new String[n];
        int[] js = new int[n];
        int ja, ts;
        double rata;

        rata = 0;
        ts = 0;
        i = 0;
        for (i = 0; i <= n - 1; i++) {
            for (i = 0; i <= n - 1; i++) {
                nm[i] = input.nextLine();
                do {
                    js[i] = Integer.parseInt(input.nextLine());
                } while (js[i] < 0 || js[i] > 7);
                ts = ts + js[i];
            }
        }
        for (i = 0; i <= n - 1; i++) {
            System.out.println(nm[i] + "=" + js[i]);
        }
        rata = (double) ts / n;
        System.out.println("Total shift=" + ts);
        System.out.println("Rata-rata=" + rata);
        b = js[0];
        d = js[0];
        for (i = 0; i <= n - 1; i++) {
            if (js[i] > b) {
                b = js[i];
            } else {
                b = b;
            }
        }
        for (i = 0; i <= n - 1; i++) {
            if (js[i] < d) {
                d = js[i];
            } else {
                d = d;
            }
        }
        System.out.println("Shift terbanyak:" + b);
        System.out.println("Shift tersedikit:" + d);
    }
}
